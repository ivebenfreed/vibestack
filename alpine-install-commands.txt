# Alpine Linux Installation Commands (Passwordless Setup)

# 1. Login as root (no password)
# 2. Run these commands step by step:

# Basic setup with minimal prompts
setup-keymap us
setup-hostname vibestack-dev

# Network setup
setup-interfaces -a

# Set empty root password (just press Enter twice)
passwd

# Timezone
setup-timezone

# Install to disk
setup-disk

# Alternative: Use response file for fully automated setup
# Create this file and run: setup-alpine -f /tmp/answerfile

cat > /tmp/answerfile << 'EOF'
KEYMAPOPTS="us us"
HOSTNAMEOPTS="-n vibestack-dev"
INTERFACESOPTS="auto lo
iface lo inet loopback

auto eth0  
iface eth0 inet dhcp
    hostname vibestack-dev
"
DNSOPTS="-d local 8.8.8.8"
TIMEZONEOPTS="-z UTC"
PROXYOPTS="none"
APKREPOSOPTS="-r"
SSHDOPTS="-c openssh"
NTPOPTS="-c chrony"
DISKOPTS="-m sys /dev/vda"
EOF

# Run automated setup
setup-alpine -f /tmp/answerfile

# After installation, reboot
reboot
