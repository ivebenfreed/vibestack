# Copy and paste these commands into the Alpine VM terminal

# 1. First, setup SSH and basic tools
apk update
apk add openssh bash curl wget git

# Enable SSH
rc-update add sshd default
service sshd start

# 2. Install development tools
apk add \
    docker docker-compose docker-cli-compose \
    nodejs npm python3 make g++ linux-headers \
    postgresql-client chromium \
    build-base libc6-compat gcompat \
    shadow sudo tmux nano

# Install pnpm
npm install -g pnpm@9.15.1

# 3. Configure Docker
rc-update add docker default
service docker start

# 4. Create developer user
adduser -D -s /bin/bash developer
echo "developer:developer" | chpasswd
adduser developer wheel
adduser developer docker
echo "developer ALL=(ALL) NOPASSWD:ALL" >> /etc/sudoers

# 5. Install XFCE desktop
apk add \
    xfce4 xfce4-terminal xfce4-screensaver \
    lightdm lightdm-gtk-greeter \
    firefox \
    dbus eudev

# 6. Configure auto-login
mkdir -p /etc/lightdm
cat > /etc/lightdm/lightdm.conf << 'EOF'
[Seat:*]
autologin-user=developer
autologin-user-timeout=0
greeter-session=lightdm-gtk-greeter
EOF

# 7. Enable services
rc-update add lightdm default
rc-update add dbus default
rc-update add udev default

# 8. Configure developer user
su - developer -c 'echo "exec startxfce4" > ~/.xinitrc && chmod +x ~/.xinitrc'

# 9. Install Claude Code for developer user
su - developer -c 'curl -fsSL https://claude.ai/install.sh | sh'

# 10. Reboot to start XFCE desktop
reboot
